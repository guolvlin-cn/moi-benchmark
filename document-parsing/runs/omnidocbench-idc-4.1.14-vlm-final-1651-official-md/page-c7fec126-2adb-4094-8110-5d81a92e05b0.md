Given the proof of Lemma A.1, it suffices to establish

            $$\mathbb{E} _ {j - 1} \left| \bar{\alpha} _ {j, r} \right| ^ {k} \lesssim \frac{(\log n) ^ {4 k}}{n ^ {k}} \mathbb{E} _ {j - 1} \bar{\Gamma} _ {j} ^ {4 k}, \quad r \in [ 4 ]. \tag{A.52}$$
For brevity, we bound only $\mathbb{E}_{j-1}|\overline{\alpha}_{j,2}|^k$ and $\mathbb{E}_{j-1}|\overline{\alpha}_{j,3}|^k$; the analysis of $\overline{\alpha}_{j,1}$ and $\overline{\alpha}_{j,4}$ does not differ substantively from the arguments presented below and in Section A.1.
(2) We expand $\bar{\alpha}_{j,2}$ using the identity $\pmb{R}_0 = \pmb{R}_{0,-j} - \pmb{R}_0(\pmb{K}_0 - \pmb{K}_{0,-j})\pmb{R}_{0,-j}$:
          $$\begin{array}{l} \bar{\alpha} _ {j, 2} = - \left(\mathbb{E} _ {j} - \mathbb{E} _ {j - 1}\right) \boldsymbol{u} ^ {\top} \boldsymbol{R} _ {0, - j} \boldsymbol{K} _ {0} ^ {(m)} \boldsymbol{R} _ {0} \left(\boldsymbol{K} _ {0} - \boldsymbol{K} _ {0, - j}\right) \boldsymbol{R} _ {0, - j} \boldsymbol{u} \\ = - \left(\mathbb{E} _ {j} - \mathbb{E} _ {j - 1}\right) \left(\bar{\delta} _ {j, 1} + \bar{\delta} _ {j, 2} - \bar{\delta} _ {j, 3}\right), \\ \end{array}$$
          $$\bar{\delta} _ {j, 1} := \boldsymbol{u} ^ {\top} \boldsymbol{R} _ {0, - j} \boldsymbol{K} _ {0, - j} ^ {(m)} \boldsymbol{R} _ {0, - j} \left(\boldsymbol{K} _ {0} - \boldsymbol{K} _ {0, - j}\right) \boldsymbol{R} _ {0, - j} \boldsymbol{u}, \tag{A.53}$$
          $$\begin{array}{l} \bar{\delta} _ {j, 2} := \boldsymbol{u} ^ {\top} \boldsymbol{R} _ {0, - j} \left(\boldsymbol{K} _ {0} ^ {(m)} - \boldsymbol{K} _ {0, - j} ^ {(m)}\right) \boldsymbol{R} _ {0} \left(\boldsymbol{K} _ {0} - \boldsymbol{K} _ {0, - j}\right) \boldsymbol{R} _ {0, - j} \boldsymbol{u}, \\ \bar{\delta} _ {j, 3} := \boldsymbol{u} ^ {\top} \boldsymbol{R} _ {0, - j} \boldsymbol{K} _ {0, - j} ^ {(m)} \boldsymbol{R} _ {0, - j} (\boldsymbol{K} _ {0} - \boldsymbol{K} _ {0, - j}) \boldsymbol{R} _ {0} (\boldsymbol{K} _ {0} - \boldsymbol{K} _ {0, - j}) \boldsymbol{R} _ {0, - j} \boldsymbol{u}. \\ \end{array}$$
  The component of $\bar{\alpha}_{j,2}$ involving $\bar{\delta}_{j,1}$ is bounded similarly to (A.26): denoting $\widetilde{\pmb{u}}_j := \pmb{R}_{0,-j}\pmb{u}$
    $$\begin{array}{l} \left| (\mathbb{E} _ {j} - \mathbb{E} _ {j - 1}) \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \boldsymbol{K} _ {0, - j} ^ {(m)} \boldsymbol{R} _ {0, - j} \boldsymbol{\Delta} _ {j, 1} \widetilde{\boldsymbol{u}} _ {j} \right| ^ {k} = \frac{1}{n ^ {k}} \left| (\mathbb{E} _ {j} - \mathbb{E} _ {j - 1}) \boldsymbol{z} _ {j} ^ {\top} \widetilde{\boldsymbol{u}} _ {j} \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \boldsymbol{K} _ {0, - j} ^ {(m)} \boldsymbol{R} _ {0, - j} \boldsymbol{z} _ {j} \right| ^ {k} \\ \lesssim \frac{1}{n ^ {k}} \mathbb{E} _ {j - 1} \| \widetilde{\boldsymbol{u}} _ {j} \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \boldsymbol{K} _ {0, - j} ^ {(m)} \boldsymbol{R} _ {0, - j} \| _ {F} ^ {k} \\ \lesssim \frac{1}{n ^ {5 k / 4}} \mathbb{E} _ {j - 1} \overline{{\Gamma}} _ {j} ^ {k}, \\ \end{array}$$
    $$\begin{array}{l} \left| \left(\mathbb{E} _ {j} - \mathbb{E} _ {j - 1}\right) \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \boldsymbol{K} _ {0, - j} ^ {(m)} \boldsymbol{R} _ {0, - j} \boldsymbol{\Delta} _ {j, 2} \widetilde{\boldsymbol{u}} _ {j} \right| ^ {k} \lesssim \frac{1}{n ^ {k / 2}} \mathbb{E} _ {j - 1} \left\| \boldsymbol{K} _ {0, - j} \left(f ^ {\prime} - a _ {1}\right) \odot \left(\widetilde{\boldsymbol{u}} _ {j} \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \boldsymbol{K} _ {0, - j} ^ {(m)}\right) \right\| _ {F} ^ {k} \tag{A.54} \\ \lesssim \frac{1}{n ^ {5 k / 4}} \mathbb{E} _ {j - 1} \overline{{\Gamma}} _ {j} ^ {2 k}, \\ \end{array}$$
        $$\mathbb{E} _ {j - 1} \left| \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \boldsymbol{K} _ {0, - j} ^ {(m)} \boldsymbol{R} _ {0, - j} \boldsymbol{\Delta} _ {j, 3} \widetilde{\boldsymbol{u}} _ {j} \right| ^ {k} \lesssim \frac{(\log n) ^ {2 L k}}{n ^ {5 k / 4}} \mathbb{E} _ {j - 1} \overline{{\Gamma}} _ {j} ^ {2 k}.$$
  Together with Lemma A.9, these bounds yield

        $$\mathbb{E} _ {j - 1} | \bar{\delta} _ {j, 1} | ^ {k} = \mathbb{E} _ {j - 1} \left| \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \boldsymbol{K} _ {0, - j} ^ {(m)} \boldsymbol{R} _ {0, - j} (\boldsymbol{\Delta} _ {j, 1} + \boldsymbol{\Delta} _ {j, 2} + \boldsymbol{\Delta} _ {j, 3}) \widetilde{\boldsymbol{u}} _ {j} \right| ^ {k} \lesssim \frac{(\log n) ^ {2 L k}}{n ^ {5 k / 4}} \mathbb{E} _ {j - 1} \overline{{\Gamma}} _ {j} ^ {2 k}. \tag{A.55}$$
      Now, we consider $\bar{\delta}_{j,2}$. By Lemma A.13 and the independence of $z_{j}$ and $(\overline{\Gamma}_j,\widetilde{\pmb{u}}_j)$,
    $$\begin{array}{l} \mathbb{E} _ {\boldsymbol{z} _ {j}} \left| \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \left(\boldsymbol{K} _ {0} ^ {(m)} - \boldsymbol{K} _ {0, - j} ^ {(m)}\right) \boldsymbol{R} _ {0} \Delta_ {j, 1} \widetilde{\boldsymbol{u}} _ {j} \right| ^ {k} \lesssim \frac{1}{n ^ {k / 2}} \mathbb{E} _ {\boldsymbol{z} _ {j}} \left| \| \boldsymbol{K} _ {0} ^ {(m)} - \boldsymbol{K} _ {0, - j} ^ {(m)} \| _ {2} \| \boldsymbol{z} _ {j} \| _ {\infty} \cdot \boldsymbol{z} _ {j} ^ {\top} \widetilde{\boldsymbol{u}} _ {j} \right| ^ {k} \\ \lesssim \frac{1}{n ^ {k}} \mathbb{E} _ {\boldsymbol{z} _ {j}} \left| (1 + \| \boldsymbol{z} _ {j} \| _ {\infty} ^ {2 (L + 1)}) \bar{\Gamma} _ {j} ^ {2} \cdot (1 + | \boldsymbol{z} _ {j} ^ {\top} \boldsymbol{v} |) \cdot \boldsymbol{z} _ {j} ^ {\top} \widetilde{\boldsymbol{u}} _ {j} \right| ^ {k} \\ \lesssim \frac{(\log n) ^ {2 (L + 1) k}}{n ^ {k}} \overline{{\Gamma}} _ {j} ^ {2 k}, \\ \end{array}$$
    $$\begin{array}{l} \mathbb{E} _ {\boldsymbol{z} _ {j}} \left| \widetilde{\boldsymbol{u}} _ {j} ^ {\top} \left(\boldsymbol{K} _ {0} ^ {(m)} - \boldsymbol{K} _ {0, - j} ^ {(m)}\right) \boldsymbol{R} _ {0} \Delta_ {j, 2} \widetilde{\boldsymbol{u}} _ {j} \right| ^ {k} \lesssim \frac{1}{n ^ {k / 2}} \mathbb{E} _ {\boldsymbol{z} _ {j}} \left| \left\| \boldsymbol{K} _ {0} ^ {(m)} - \boldsymbol{K} _ {0, - j} ^ {(m)} \right\| _ {2} \left\| \boldsymbol{K} _ {0, - j} (f ^ {\prime} - a _ {1}) \right\| _ {2} \| \boldsymbol{z} _ {j} \| _ {\infty} ^ {2} \right| ^ {k} \\ \lesssim \frac{(\log n) ^ {(2 L + 3) k}}{n ^ {k}} \bar{\Gamma} _ {j} ^ {3 k}. \\ \end{array}$$
