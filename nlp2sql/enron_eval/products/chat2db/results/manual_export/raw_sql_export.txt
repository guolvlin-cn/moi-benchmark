FROM (

    AVG(COALESCE(rc.recipient_count, 0)) AS avg_recipients
  FROM enron_emailinfo ei

  GROUP BY sender

ORDER BY avg_recipients DESC;

-- m14: 哪些邮件的标准To收件地址数量与X-To原始收件人显示值数量差异最大？
SELECT ei.id,
  COALESCE((SELECT COUNT(DISTINCT et.`to`) FROM enron_emailto et WHERE et.id = ei.id), 0) AS to_count,
  COALESCE((SELECT COUNT(DISTINCT TRIM(xt.xto)) FROM enron_emailxto xt WHERE xt.id = ei.id AND xt.xto IS NOT NULL AND TRIM(xt.xto) != ''), 0) AS xto_count,
  ABS(COALESCE((SELECT COUNT(DISTINCT et.`to`) FROM enron_emailto et WHERE et.id = ei.id), 0)
    - COALESCE((SELECT COUNT(DISTINCT TRIM(xt.xto)) FROM enron_emailxto xt WHERE xt.id = ei.id AND xt.xto IS NOT NULL AND TRIM(xt.xto) != ''), 0)) AS abs_diff,
  LEFT(ei.`to`, 200) AS to_preview,
  LEFT(ei.xto, 200) AS xto_preview
FROM enron_emailinfo ei
ORDER BY abs_diff DESC, ei.id ASC
LIMIT 10;

-- h04: 对每个非空归档员工目录people，列出每个people的前3个发件地址、邮件数和名次。
SELECT people, sender, cnt, rn
FROM (
  SELECT e.people, TRIM(LOWER(ei.`from`)) AS sender,
    COUNT(DISTINCT e.id) AS cnt,
    ROW_NUMBER() OVER (PARTITION BY e.people
      ORDER BY COUNT(DISTINCT e.id) DESC, TRIM(LOWER(ei.`from`)) ASC) AS rn
  FROM enron_email e
  JOIN enron_emailinfo ei ON e.id = ei.id
  WHERE e.people IS NOT NULL AND TRIM(e.people) != ''
    AND ei.`from` IS NOT NULL AND TRIM(ei.`from`) != ''
  GROUP BY e.people, sender
) t
WHERE rn <= 3
ORDER BY people ASC, rn ASC;

-- h05: 邮件往来最忙的是哪一天？那天发邮件最多的5个邮箱是谁？
WITH busiest_day AS (
  SELECT DATE(STR_TO_DATE(SUBSTRING_INDEX(`date`, ' ', 5), '%a, %d %b %Y')) AS send_date
  FROM enron_emailinfo WHERE `date` IS NOT NULL AND `date` != ''
  GROUP BY send_date ORDER BY COUNT(DISTINCT id) DESC LIMIT 1
)
SELECT TRIM(LOWER(`from`)) AS sender, COUNT(DISTINCT id) AS cnt
FROM enron_emailinfo
WHERE `date` IS NOT NULL AND `date` != ''
  AND DATE(STR_TO_DATE(SUBSTRING_INDEX(`date`, ' ', 5), '%a, %d %b %Y')) = (SELECT send_date FROM busiest_day)
  AND `from` IS NOT NULL AND TRIM(`from`) != ''
GROUP BY sender ORDER BY cnt DESC LIMIT 5;

-- h06: 将每封邮件按规范化发件地址分类internal和external，统计不同邮件数及其百分比。
SELECT type, cnt,
  ROUND(cnt * 100.0 / (
    SELECT COUNT(DISTINCT id) FROM enron_emailinfo
    WHERE `from` IS NOT NULL AND TRIM(`from`) != ''
  ), 2) AS pct
FROM (
  SELECT CASE WHEN TRIM(LOWER(`from`)) LIKE '%@enron.com' THEN 'internal' ELSE 'external' END AS type,
    COUNT(DISTINCT id) AS cnt
  FROM enron_emailinfo
  WHERE `from` IS NOT NULL AND TRIM(`from`) != ''
  GROUP BY CASE WHEN TRIM(LOWER(`from`)) LIKE '%@enron.com' THEN 'internal' ELSE 'external' END
) t
ORDER BY type ASC;

-- h07: 仅考虑campbell-l、allen-p和badeer-r，找出至少出现在其中两个不同people集合中的地址。
SELECT email, COUNT(DISTINCT people) AS people_count
FROM (
  SELECT DISTINCT TRIM(LOWER(ei.`from`)) AS email, e.people
  FROM enron_emailinfo ei JOIN enron_email e ON ei.id = e.id
  WHERE e.people IN ('campbell-l','allen-p','badeer-r')
    AND ei.`from` IS NOT NULL AND TRIM(ei.`from`) != ''
  UNION
  SELECT DISTINCT TRIM(LOWER(et.`to`)) AS email, e.people
  FROM enron_emailto et JOIN enron_email e ON et.id = e.id
  WHERE e.people IN ('campbell-l','allen-p','badeer-r')
    AND et.`to` IS NOT NULL AND TRIM(et.`to`) != ''
) t
GROUP BY email
HAVING people_count >= 2
ORDER BY people_count DESC, email ASC;

-- h08: Larry Campbell和Phillip Allen都给哪些邮箱发过邮件？分别发了多少封？
SELECT et.`to`,
  SUM(CASE WHEN ei.`from` = 'larry.campbell@enron.com' THEN 1 ELSE 0 END) AS campbell_count,
  SUM(CASE WHEN ei.`from` = 'phillip.allen@enron.com' THEN 1 ELSE 0 END) AS allen_count
FROM enron_emailto et
JOIN enron_emailinfo ei ON et.id = ei.id
WHERE ei.`from` IN ('larry.campbell@enron.com', 'phillip.allen@enron.com')
GROUP BY et.`to`
HAVING campbell_count > 0 AND allen_count > 0
ORDER BY campbell_count DESC, allen_count DESC;

-- h09: 对enron_emailorig中存在记录的每封当前邮件，统计历史邮件头数量指标。
SELECT
  ROUND(AVG(orig_count), 4) AS avg_orig_count,
  MAX(orig_count) AS max_orig_count,
  SUM(CASE WHEN orig_count >= 3 THEN 1 ELSE 0 END) AS ge3_count
FROM (
  SELECT eo.id, COUNT(*) AS orig_count
  FROM enron_emailorig eo
  GROUP BY eo.id
) t;

-- h10: 每个人邮件最多的文件夹是哪个？并列的也都列出来。
SELECT people, mailbox, cnt
FROM (
  SELECT people, mailbox, COUNT(DISTINCT id) AS cnt,
    RANK() OVER (PARTITION BY people ORDER BY COUNT(DISTINCT id) DESC) AS rn
